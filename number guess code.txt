print("Think of a number between 1 to 100")

low = 1
high = 100

for step in range(1, 9):
    guess = (low + high) // 2

    print("Step", step)
    print("My guess is:", guess)

    user = input("Enter yes / greater / lesser: ").lower()

    if user == "yes":
        print("Yay! I guessed the correct number!")
        break

    elif user == "greater":
        low = guess + 1

    elif user == "lesser":
        high = guess - 1

    else:
        print("Invalid input! Try again.")
        continue

else:
    print("I couldn't guess your number in 8 steps!")
